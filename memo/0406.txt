예외처리
try{
  예외가 발생할 수도 있는 코드
} catch(매개변수){
  예외가 발생하면 실행할 코드
} catch(매개변수){
  예외가 발생하면 실행할 코드
} catch(매개변수){
  예외가 발생하면 실행할 코드
} finally{
  예외가 발생하던 발생하지 않던 무조건 실행되는 코드
}
